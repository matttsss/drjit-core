# White-box test of the bit fields encoding the JIT backend (private headers)

target_link_libraries(test_backend_bits PRIVATE drjit-core)
target_include_directories(test_backend_bits PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/../src
  ${CMAKE_CURRENT_SOURCE_DIR}/../ext/robin_map/include
  ${CMAKE_CURRENT_SOURCE_DIR}/../ext/nanothread/include
  ${CMAKE_CURRENT_SOURCE_DIR}/../ext/lz4)

  target_compile_definitions(test_backend_bits PRIVATE -DDRJIT_ENABLE_VULKAN=1)
endif()
add_test(NAME test_backend_bits COMMAND test_backend_bits
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")

if (DRJIT_ENABLE_VULKAN)
  # Dynamic loading of the Vulkan loader and SPIRV-Tools (private headers).
  # Skipped at runtime if they are unavailable, unless the environment variable
  # DRJIT_TEST_REQUIRE_VULKAN=1 is set.
  add_executable(test_vulkan_api vulkan_api.cpp)
  set_property(TARGET test_vulkan_api PROPERTY CXX_STANDARD 17)
  target_link_libraries(test_vulkan_api PRIVATE drjit-core)
  target_include_directories(test_vulkan_api PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../src)
  target_compile_definitions(test_vulkan_api PRIVATE -DDRJIT_ENABLE_VULKAN=1)
  if (DRJIT_DYNAMIC_VULKAN)
    target_compile_definitions(test_vulkan_api PRIVATE -DDRJIT_DYNAMIC_VULKAN=1)
  else()
    target_include_directories(test_vulkan_api PRIVATE ${DRJIT_VULKAN_INCLUDE_DIR})
  endif()
  add_test(NAME test_vulkan_api COMMAND test_vulkan_api
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")

  # Compile-time comparison of the self-contained declarations in
  # 'src/vulkan_api.h' against the official headers. 'vulkan_layout.cpp' is
  # generated by 'vulkan_layout_gen.py' and does not need the Dr.Jit library.
  if (DRJIT_VULKAN_INCLUDE_DIR)
    find_path(DRJIT_SPIRV_TOOLS_INCLUDE_DIR NAMES spirv-tools/libspirv.h
      HINTS "$ENV{VULKAN_SDK}/include" "$ENV{VULKAN_SDK}/Include"
      DOC "Directory containing the SPIRV-Tools headers (spirv-tools/libspirv.h)")
    mark_as_advanced(DRJIT_SPIRV_TOOLS_INCLUDE_DIR)

    add_executable(test_vulkan_layout vulkan_layout.cpp)
    set_property(TARGET test_vulkan_layout PROPERTY CXX_STANDARD 17)
    target_include_directories(test_vulkan_layout PRIVATE
      ${CMAKE_CURRENT_SOURCE_DIR}/../src ${DRJIT_VULKAN_INCLUDE_DIR})
    if (DRJIT_SPIRV_TOOLS_INCLUDE_DIR)
      target_include_directories(test_vulkan_layout PRIVATE
        ${DRJIT_SPIRV_TOOLS_INCLUDE_DIR})
      target_compile_definitions(test_vulkan_layout PRIVATE
        -DDRJIT_CHECK_SPIRV_TOOLS=1)
    else()
      message(STATUS "Dr.Jit-Core: SPIRV-Tools headers not found, "
        "test_vulkan_layout will only check the Vulkan declarations.")
    endif()
    add_test(NAME test_vulkan_layout COMMAND test_vulkan_layout
      WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  else()
    message(STATUS "Dr.Jit-Core: official Vulkan headers not found, "
      "skipping test_vulkan_layout.")
  endif()
endif()

foreach (TEST_FILE ${TEST_FILES})
  get_filename_component(TEST_NAME ${TEST_FILE} NAME_WE)
  add_executable(test_${TEST_NAME} ${TEST_NAME}.cpp test.h test.cpp ${TEST_NAME}.cpp)
  target_link_libraries(test_${TEST_NAME} PRIVATE drjit-core)
  target_compile_definitions(test_${TEST_NAME} PRIVATE -DTEST_NAME="${TEST_NAME}")
  target_compile_features(test_${TEST_NAME} PRIVATE cxx_std_11)

  if (DRJIT_ENABLE_OPTIX)
    target_compile_definitions(test_${TEST_NAME} PRIVATE -DDRJIT_ENABLE_OPTIX=1)
  endif()
  add_test(
    NAME ${TEST_NAME}
    COMMAND test_${TEST_NAME}
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  )
endforeach()
